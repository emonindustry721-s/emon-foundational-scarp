Starter Code Snippets

port_scanner.py

import socket

def port_scan(ip, start_port, end_port):
    open_ports = []
    for port in range(start_port, end_port + 1):
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(1)
        result = sock.connect_ex((ip, port))
        if result == 0:
            open_ports.append(port)
        sock.close()
    return open_ports

if __name__ == "__main__":
    print(port_scan("127.0.0.1", 1, 1000))

header_checker.py

import requests

def check_headers(url):
    headers = requests.get(url).headers
    checks = {
        "Content-Security-Policy": "CSP",
        "X-Frame-Options": "X-Frame",
        "Strict-Transport-Security": "HSTS"
    }
    report = {name: (header in headers) for header, name in checks.items()}
    return report

if __name__ == "__main__":
    print(check_headers("https://example.com"))

log_analyzer.raku

use v6;

my %failed;
for 'access.log'.IO.lines -> $line {
    if $line ~~ / "Failed login" .* "from " (\S+)/ {
        %failed{$0}++;
    }
}

for %failed.kv -> $ip, $count {
    say "$ip,$count" if $count > 5;
}
